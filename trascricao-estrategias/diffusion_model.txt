+ Distribuição conjunta coerente (High/Low/Close consistentes);
+ Fácil obter prob. de eventos (gap, stop-first, etc.);

− Se dados forem magros, pode “alucinar” regimes raros (regularize!).

import torch, torch.nn as nn, torch.nn.functional as F

def timestep_embed(t, dim):
    half = dim//2
    freqs = torch.exp(torch.arange(half, device=t.device)*(-torch.log(torch.tensor(10000.0))/half))
    ang = t[:,None]*freqs[None]
    return torch.cat([torch.sin(ang), torch.cos(ang)], dim=1)

class CondBlock(nn.Module):
    def __init__(self, d, d_cond): super().__init__(); self.fc = nn.Linear(d_cond, d*2)
    def forward(self, h, c):
        g, b = self.fc(c).chunk(2, dim=-1)   # FiLM
        return h*torch.sigmoid(g) + b

class EpsModel(nn.Module):
    def __init__(self, d_y, d_cond, d=256, n=4):
        super().__init__()
        self.inp = nn.Linear(d_y, d)
        self.time = nn.Linear(128, d)
        self.cond = CondBlock(d, d_cond)
        self.blocks = nn.ModuleList([nn.TransformerEncoderLayer(d, 4, batch_first=True) for _ in range(n)])
        self.out = nn.Linear(d, d_y)

    def forward(self, y_noisy, t_idx, cond_vec):
        h = self.inp(y_noisy) + self.time(timestep_embed(t_idx, 128))
        h = self.cond(h, cond_vec[:,None,:].expand_as(h))
        for blk in self.blocks: h = blk(h)
        return self.out(h)

# Treino DDPM condicional (y: [B,H,d_y], cond_vec: [B,d_cond])
